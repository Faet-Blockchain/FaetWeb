What a correct “Disconnect” must do
Stop listening to provider events.

Clear your app state (account, chainId, flags, any “autoconnect” markers).

Try to revoke eth_accounts permission (wallet may ignore).

Don’t auto-reconnect on mount (use eth_accounts for silent check; only call eth_requestAccounts on Connect click).

// helper
const getEth = () =>
  (typeof window !== "undefined" ? window.ethereum : undefined) ?? null;

// listener refs so we can remove the same function references
const onAccountsChanged = (accs: string[]) => {
  setAccount(accs?.[0] ?? null);
};
const onChainChanged = (cid: string) => {
  setCurrentChainId(cid);
};
const onDisconnect = () => {
  setAccount(null);
  setCurrentChainId(null);
  setWrongNetwork(false);

// attach & cleanup (where you currently add listeners)
useEffect(() => {
  const eth = getEth();
  if (!eth) return;

  eth.on?.("accountsChanged", onAccountsChanged);
  eth.on?.("chainChanged", onChainChanged);
  eth.on?.("disconnect", onDisconnect);

  return () => {
    eth.removeListener?.("accountsChanged", onAccountsChanged);
    eth.removeListener?.("chainChanged", onChainChanged);
    eth.removeListener?.("disconnect", onDisconnect);
  };
}, []);

// disconnect button handler (replace your current one)
const disconnect = async () => {
  const eth = getEth();

  // 1) remove listeners
  eth?.removeListener?.("accountsChanged", onAccountsChanged);
  eth?.removeListener?.("chainChanged", onChainChanged);
  eth?.removeListener?.("disconnect", onDisconnect);

  // 2) try to revoke permission (supported in some wallets)
  try {
    await eth?.request?.({
      method: "wallet_revokePermissions",
      params: [{ eth_accounts: {} }],
    });
  } catch { /* ignore if unsupported */ }

  // 3) clear app state
  setAccount(null);
  setCurrentChainId(null);
  setWrongNetwork(false);
  setIsConnecting(false);
  localStorage.removeItem("faet:lastConnectedWallet");

  // 4) verify current authorization for UI hints
  try {
    const accs = (await eth?.request?.({ method: "eth_accounts" })) as string[] | undefined;
    if (accs?.length) {
      console.warn("Still authorized in MetaMask; user must disconnect in wallet UI (Connected sites).");
    }
  } catch { /* noop */ }
};
